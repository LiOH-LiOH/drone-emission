\block{Background}{
  \textbf{Drone delivery} is a transformative technology for last-mile logistics, with the global market projected to grow from \$3.47B (2025) to \$20.98B (2034) at a CAGR of \textbf{19.45\%} \parencite{fortunebi2026drone}.

  \vspace{0.5em}
  \textbf{Food delivery} is an ideal application due to:
  \begin{itemize}[leftmargin=1.5em, itemsep=0.2em]
    \item Lighter packages and shorter distances ($<$3 km)
    \item High time sensitivity matching drone capabilities
    \item Massive market: \textbf{545 million} users in China
  \end{itemize}

  \vspace{0.5em}
  \begin{tikzpicture}
    \node[draw=colorThree, fill=colorTwo, rounded corners=10pt, inner sep=15pt, text width=0.9\linewidth, align=left] {
      \textbf{Research Gap:} Most existing studies \textbf{lack real origin-destination (OD) data} and rely on aggregate metrics that cannot capture route-level emission variations.
    };
  \end{tikzpicture}
}

\block{Contributions}{
  \begin{enumerate}[leftmargin=1.5em, itemsep=0.3em]
    \item A \textbf{gravity-based demand generation method} producing door-to-door order data for precise emission analysis.
    \item The \textbf{first comprehensive comparison} of carbon emissions between e-bike and drone delivery across multiple Chinese cities.
    \item Identification of \textbf{key urban features} that determine when drone delivery achieves lower emissions.
  \end{enumerate}
}

\block{OD Generation Pipeline}{
  \begin{center}
    \begin{tikzpicture}[
        node distance=0.8cm and 1.8cm,
        data/.style={rectangle, draw=blue!60, fill=blue!10, minimum width=2.4cm, minimum height=1cm, align=center, font=\small},
        processed/.style={rectangle, draw=orange!70, fill=orange!15, rounded corners=5pt, minimum width=2.4cm, minimum height=1cm, align=center, font=\small},
        output/.style={rectangle, draw=gray!70, fill=gray!10, rounded corners=5pt, minimum width=2.4cm, minimum height=1cm, align=center, font=\small},
        model/.style={diamond, draw=teal!70, fill=teal!15, minimum width=1.5cm, minimum height=1.2cm, align=center, font=\small, inner sep=2pt},
        arrow/.style={-{Stealth[scale=0.8]}, thick, draw=gray!70}
      ]
      \node[data] (sales) {Monthly Sales\\(1.07M merchants)};
      \node[data, right=2cm of sales] (building) {Building\\Footprint};
      \node[processed, below=1cm of sales] (merchant) {Merchant\\Locations};
      \node[processed, below=1cm of building] (buildings) {Nearby\\Buildings};
      \path (merchant) -- (buildings) coordinate[midway] (mid);
      \node[model, below=1cm of mid] (gravity) {Gravity\\Model};
      \node[output, below=1cm of gravity] (od) {OD Matrix};
      \draw[arrow] (sales) -- (merchant);
      \draw[arrow] (building) -- (buildings);
      \draw[arrow] (gravity) -- (od);
      \draw[arrow] (merchant.south) -- (gravity.north west);
      \draw[arrow] (buildings.south) -- (gravity.north east);
    \end{tikzpicture}
  \end{center}

  \vspace{0.3em}
  \textbf{Gravity model} allocates deliveries based on building volume and distance:
  \[
    D_{ij} = P_{i} \cdot \frac{v_{j}^\alpha \,/\, d_{ij}^{\beta}}{\sum_{k \in \mathcal{D}} v_{k}^\alpha \,/\, d_{ik}^{\beta}}
  \]
  where $P_i$ is merchant demand, $v_j$ is building volume, $d_{ij}$ is distance, and $\alpha{=}1.7$, $\beta{=}0.5$ (calibrated to mean delivery distance of 2.23 km \parencite{zhang_exploration_2023}).
}

\block{Emission Models}{
  \textbf{E-bike power model} \parencite{kirschstein_comparison_2020}:
  \vspace{-0.3em}
  \[ P_{E} = P_{\text{rolling}} + P_{\text{air}} + P_{\text{grade}} + P_{\text{acc}} \]

  \textbf{Drone power model} (multi-rotor) \parencite{kirschstein_comparison_2020}:
  \vspace{-0.3em}
  \[ P_{U} = P_{\text{induced}} + P_{\text{profile}} + P_{\text{drag}} + P_{\text{climb}} + P_{\text{aux}} \]

  \textbf{Lifecycle analysis} includes production, use-phase, and disposal emissions \parencite{figliozzi_lifecycle_2017}, allocated per delivery based on vehicle lifetime capacity.
}

\block{Key Findings}{
  \begin{tikzpicture}
    \node[draw=colorThree, fill=colorTwo, rounded corners=10pt, inner sep=15pt, text width=0.92\linewidth, align=left] {
      \begin{itemize}[leftmargin=1.2em, itemsep=0.3em]
        \item All-drone delivery reduces emissions by \textbf{27\%} vs.\ all e-bike.
        \item Optimal drone--e-bike mix achieves \textbf{30\%} reduction.
        \item \textbf{Route factor} (road detour) is the strongest predictor of drone emission advantage.
        \item Drone benefits increase with \textbf{road network complexity} and \textbf{delivery distance}.
        \item Results are robust across diverse urban environments in China.
      \end{itemize}
    };
  \end{tikzpicture}
}
