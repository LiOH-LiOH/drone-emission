  \block{Research Objectives}{
    \begin{enumerate}[leftmargin=1.2em, itemsep=0.15em, topsep=0.1em]
      \item Develop a \textbf{gravity-based food delivery demand generation method} producing door-to-door order data for precise emission analysis.
      \item Conduct the \textbf{first comprehensive comparison} of carbon emissions between e-bike and drone delivery systems across multiple Chinese cities.
      \item Identify \textbf{key urban features} that determine when drones achieve lower emissions, providing actionable insights for sustainable logistics planning.
    \end{enumerate}
  }

      \textbf{Emission Reduction Potential}
      \begin{itemize}[leftmargin=1.2em, itemsep=0.15em]
        \item Threshold-based assignment: orders above distance $d^*$ go to drones, below to e-bikes. Even with e-bike bundling ($\kappa{=}0.7$--$0.9$), optimal drone deployment achieves significant emission savings.
      \end{itemize}

      \vspace{0.5em}
      \textbf{Key Findings}
      \begin{itemize}[leftmargin=1.2em, itemsep=0.15em]
        \item All-drone delivery reduces emissions by \textbf{33\%} on average vs.\ all e-bike; optimal drone--e-bike mix achieves \textbf{36\%} mean reduction.
        \item Results are robust across both city-level (Shenzhen) and cross-region analyses.
        \item \textbf{Route factor} (detour ratio) is the strongest predictor of drone emission advantage, with benefits increasing alongside road network complexity and delivery distance.
      \end{itemize}

  \block{Future Work}{
    \begin{itemize}[leftmargin=1.2em, itemsep=0.15em]
      \item Scale to \textbf{all major Chinese cities} and, ultimately, nationwide coverage.
      \item Incorporate \textbf{economic analysis} combining emission and cost models.
      \item Account for \textbf{weather impacts} (wind, rain) on drone energy consumption.
      \item Extend to \textbf{other delivery modes} (e.g., gasoline cars, electric cars, autonomous vehicles) and compare emissions across last-mile scenarios.
      \item Develop a \textbf{predictive model} for optimal drone deployment ratios across urban typologies, enabling emission savings estimation without full simulation.
    \end{itemize}

    \vspace{0.5em}
    % --- Step 1: Data & OD Generation ---
    \textbf{Step 1: Generating realistic delivery orders.}\enspace
    We reconstruct OD pairs using a \textbf{gravity model} with Tanner impedance \parencite{tanner1961factors}. Monthly sales of \textbf{1.07 million merchants} provide store locations; building footprints give customer locations. Each merchant $i$ with demand $P_i$ distributes orders to buildings $j$:
    \[
      D_{ij} = P_{i} \cdot \frac{v_{j}^\alpha \cdot f(d_{ij})}{\sum_{k \in \mathcal{D}} v_{k}^\alpha \cdot f(d_{ik})}
      \qquad\qquad
      f(d) = d^{\beta} \cdot e^{-\gamma \cdot d}
    \]

    \vspace{0.2em}
    \textit{Two-stage calibration.}\enspace
    \textbf{Stage 1:} We fit the Tanner impedance $f(d)$ to \textbf{278k Eleme orders} (29 days, Beijing), obtaining $\beta{=}1.113$, $\gamma{=}1.412$ ($R^2{=}0.985$).
    \textbf{Stage 2:} With $\beta, \gamma$ fixed, we sweep the volume exponent $\alpha$ until the model's mean delivery distance matches the observed 2.23\,km \parencite{xie_environmental_2021}, yielding $\alpha{=}1.7$.

    \vspace{0.2em}
    \textbf{Step 2: Computing delivery distances.}\enspace
    For each OD pair we compute: \textbf{air distance} (drone) and \textbf{road-network shortest path} (e-bike, via OSM). Their ratio is the \textbf{route factor}.

    \vspace{0.2em}
    % --- Step 3: Emission Models ---
    \textbf{Step 3: Lifecycle emission modelling} \parencite{kirschstein_comparison_2020, figliozzi_lifecycle_2017}\textbf{.}
    \begin{itemize}[leftmargin=1.5em, itemsep=0.1em]
      \item \textbf{Use phase:} physics-based power models for e-bike (rolling resistance, drag, grade) and drone (induced lift, blade drag, climb). Energy multiplied by grid emission factor yields CO$_2$e.
        \begin{itemize}[leftmargin=1.2em, itemsep=0em, topsep=0em]
          \item \textit{E-bike:} rolling + aero drag + grade + acceleration
          \item \textit{Drone:} induced lift + profile drag + parasitic + climb + auxiliary
        \end{itemize}
      \item \textbf{Production \& disposal:} body frame manufacturing, battery pack production, and end-of-life recycling and landfill; embodied emissions amortised over \textbf{lifetime delivery count}.
      \item \textbf{Comparison:} drones gain an advantage at longer distances and higher route factors due to lower per-km energy and direct flight paths.
    \end{itemize}
